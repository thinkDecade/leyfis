pub fn gate(
  ctx: Context<Gate>,
  vault_ix_data: Vec<u8>,
  audit_nonce: u64,
) -> Result<()> {

  // 1. Emergency pause check
  require(!vc.paused, GatePaused);

  // 2–4. Load and validate attestation
  let att = load_attestation(
    &ctx.accounts.attestation
  )?;

  // 5. Issuer must be in trusted list
  require(
    vc.trusted_issuers.contains(&att.issuer),
    UntrustedIssuer
  );

  // 6. Tier check
  require(
    att.tier >= vc.min_tier,
    TierInsufficient
  );

  // ✓ All checks passed. CPI to vault
  invoke(&ix, &remaining_accounts)?;

  // Always write audit entry
  set_audit_entry(
    ..., GateOutcome::Approved, 0
  );
  Ok(())
}
